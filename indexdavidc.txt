let num1 = Number(prompt("Ingresa un número"));
let num2 = Number(prompt("Ingresa otro número"));
let operacion = Number(prompt("1 suma, 2 resta, 3 multiplicación, 4 división"));

function calculadora(operacion, num1, num2) {
  let resultado;

  switch (operacion) {
    case 1:
      resultado = num1 + num2;
      console.log(`Resultado de la suma: ${resultado}`);
      break;
    case 2:
      resultado = num1 - num2;
      console.log(`Resultado de la resta: ${resultado}`);
      break;
    case 3:
      resultado = num1 * num2;
      console.log(`Resultado de la multiplicación: ${resultado}`);
      break;
    case 4:
      if (num1 || num2 === 0) {
        console.log("Error: no se puede dividir entre cero");
      } else {
        resultado = num1 / num2;
        console.log(`Resultado de la división: ${resultado}`);
      }
      break;
    default:
      console.log("Operación no válida. Elige un número del 1 al 4.");
  }
}

// Llamada a la función
calculadora(operacion, num1, num2);
